X API क्लाइंट के कॉन्फ़िगरेशन विकल्प

<div id="properties">
  ## गुण
</div>

<ResponseField name="baseUrl" type="string">
  API अनुरोधों के लिए बेस URL
</ResponseField>

<ResponseField name="bearerToken" type="string">
  प्रमाणीकरण के लिए बेयरर टोकन
</ResponseField>

<ResponseField name="accessToken" type="string">
  OAuth2 एक्सेस टोकन
</ResponseField>

<ResponseField name="oauth1" type="any">
  प्रमाणीकरण के लिए OAuth1 इंस्टेंस
</ResponseField>

<ResponseField name="headers" type="Record<string, string>">
  अनुरोधों में शामिल किए जाने वाले कस्टम हेडर
</ResponseField>

<ResponseField name="timeout" type="number">
  मिलीसेकंड में अनुरोध का टाइमआउट
</ResponseField>

<ResponseField name="retry" type="boolean">
  क्या विफल अनुरोधों को स्वचालित रूप से फिर से आज़माना है
</ResponseField>

<ResponseField name="maxRetries" type="number">
  पुनः प्रयासों की अधिकतम संख्या
</ResponseField>
