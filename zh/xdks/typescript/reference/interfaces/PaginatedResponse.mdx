分页响应接口

表示 X API 分页响应的结构。

<div id="type-parameters">
  ## 类型参数
</div>

| 名称 | 描述 |
| :------ | :------ |
| `T` | 响应中各项的类型 |

<div id="properties">
  ## 属性
</div>

<ResponseField name="data" type="T[]" required>
当前页中的条目数组

</ResponseField>

<ResponseField name="meta" type="Object">
分页元数据

<Expandable title="属性">
<ResponseField name="名称" type="类型 | 描述" required>
</ResponseField>

<ResponseField name="resultCount" type="number | 当前页的结果数量">
</ResponseField>

<ResponseField name="nextToken" type="string | 用于获取下一页的令牌">
</ResponseField>

<ResponseField name="previousToken" type="string | 用于获取上一页的令牌">
</ResponseField>
</Expandable>
</ResponseField>

<ResponseField name="includes" type="Record&lt;string, any&gt;">
附加的包含对象（用户、推文等）

</ResponseField>

<ResponseField name="errors" type="any[]">
响应中出现的任何错误

</ResponseField>
