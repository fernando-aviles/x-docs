Options de configuration du client X API

<div id="properties">
  ## Propriétés
</div>

<ResponseField name="baseUrl" type="string">
URL de base pour les requêtes à l'API

</ResponseField>

<ResponseField name="bearerToken" type="string">
Bearer Token pour l'authentification

</ResponseField>

<ResponseField name="accessToken" type="string">
Jeton d'accès OAuth2

</ResponseField>

<ResponseField name="oauth1" type="any">
Instance OAuth1 pour l'authentification

</ResponseField>

<ResponseField name="headers" type="Record&lt;string, string&gt;">
En-têtes personnalisés à inclure dans les requêtes

</ResponseField>

<ResponseField name="timeout" type="number">
Délai d'expiration de la requête en millisecondes

</ResponseField>

<ResponseField name="retry" type="boolean">
Indique s'il faut relancer automatiquement les requêtes ayant échoué

</ResponseField>

<ResponseField name="maxRetries" type="number">
Nombre maximal de tentatives de réessai

</ResponseField>
