Interfaz de respuesta paginada

Representa la estructura de una respuesta paginada de la X API.

<div id="type-parameters">
  ## Parámetros de tipo
</div>

| Nombre | Descripción |
| :------ | :------ |
| `T` | El tipo de elementos de la respuesta |

<div id="properties">
  ## Propiedades
</div>

<ResponseField name="data" type="T[]" required>
Array de elementos en la página actual

</ResponseField>

<ResponseField name="meta" type="Object">
Metadatos de paginación

<Expandable title="propiedades">
<ResponseField name="Nombre" type="Tipo | Descripción" required>
</ResponseField>

<ResponseField name="resultCount" type="number | Número de resultados en la página actual">
</ResponseField>

<ResponseField name="nextToken" type="string | Token para obtener la página siguiente">
</ResponseField>

<ResponseField name="previousToken" type="string | Token para obtener la página anterior">
</ResponseField>
</Expandable>
</ResponseField>

<ResponseField name="includes" type="Record&lt;string, any&gt;">
Objetos adicionales incluidos (usuarios, tuits, etc.)

</ResponseField>

<ResponseField name="errors" type="any[]">
Cualquier error en la respuesta

</ResponseField>
