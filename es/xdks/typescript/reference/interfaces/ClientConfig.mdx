Opciones de configuración del cliente de X API

<div id="properties">
  ## Propiedades
</div>

<ResponseField name="baseUrl" type="string">
URL base para las solicitudes de API

</ResponseField>

<ResponseField name="bearerToken" type="string">
Bearer Token para la autenticación

</ResponseField>

<ResponseField name="accessToken" type="string">
Token de acceso OAuth2

</ResponseField>

<ResponseField name="oauth1" type="any">
Instancia de OAuth1 para la autenticación

</ResponseField>

<ResponseField name="headers" type="Record&lt;string, string&gt;">
Encabezados personalizados que se incluirán en las solicitudes

</ResponseField>

<ResponseField name="timeout" type="number">
Tiempo de espera de la solicitud en milisegundos

</ResponseField>

<ResponseField name="retry" type="boolean">
Indica si se deben reintentar automáticamente las solicitudes fallidas

</ResponseField>

<ResponseField name="maxRetries" type="number">
Número máximo de reintentos

</ResponseField>
